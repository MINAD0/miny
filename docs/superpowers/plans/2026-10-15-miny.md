# miny Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build miny, a Windows Electron + React + TypeScript assistant that transcribes meetings and interviews, suggests answers through a user-chosen LLM provider, and keeps notes.

**Architecture:** Electron main process owns audio orchestration, SQLite storage, provider calls, secrets, and tracking. A preload script exposes a typed `window.miny` API. The renderer is React with two windows: a main window and a transparent always-on-top overlay. Providers and transcription sit behind interfaces so OpenAI, Anthropic, and OpenAI-compatible endpoints (OpenRouter, Ollama) are swappable.

**Tech Stack:** Electron, electron-vite, React 18+, TypeScript, Vite, better-sqlite3 (FTS5), Vitest, Playwright (Electron), Node `fetch`.

**Spec:** `docs/superpowers/specs/2026-10-15-miny-design.md`

## Global Constraints

- Target platform: Windows 10/11 x64 only in v1.
- The overlay must never call `setContentProtection`. Task 13 pins this with a test.
- API keys are stored only via Electron `safeStorage`, never in SQLite or plain config files.
- No third-party analytics SDK. Tracking is local-first; remote upload is off by default.
- A network kill switch blocks all outbound provider and tracking requests.
- Speech-to-text provider is configured separately from the LLM provider.
- Every provider call goes through the registry in Task 8. No module calls a vendor API directly.

## Review Focus

1. Empty or whitespace-only API key: clear error, no network call. Owned by Task 8.
2. Provider returns HTTP 429 repeatedly: at most 3 retries, then a visible error. Owned by Task 8.
3. Audio device removed mid-session: session stops, partial transcript is saved as a note. Owned by Task 10.
4. Note title containing `/`, `:`, emoji, or 500+ characters: exported filename is valid on Windows. Owned by Task 5.
5. Vision requested with a non-vision model: Vision is disabled and the reason is shown. Owned by Task 14.

```ts
import Database from 'better-sqlite3'

const MIGRATIONS: string[] = [
  `
  CREATE TABLE folders (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    parent_id INTEGER REFERENCES folders(id) ON DELETE SET NULL
  );
  CREATE TABLE sessions (
    id INTEGER PRIMARY KEY,
    started_at INTEGER NOT NULL,
    ended_at INTEGER,
    mode TEXT NOT NULL,
    provider TEXT,
    model TEXT
  );
  CREATE TABLE notes (
    id INTEGER PRIMARY KEY,
    title TEXT NOT NULL DEFAULT '',
    body_md TEXT NOT NULL DEFAULT '',
    folder_id INTEGER REFERENCES folders(id) ON DELETE SET NULL,
    favorite INTEGER NOT NULL DEFAULT 0,
    archived INTEGER NOT NULL DEFAULT 0,
    session_id INTEGER REFERENCES sessions(id) ON DELETE SET NULL,
    created_at INTEGER NOT NULL,
    updated_at INTEGER NOT NULL
  );
  CREATE TABLE tags (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL UNIQUE
  );
  CREATE TABLE note_tags (
    note_id INTEGER NOT NULL REFERENCES notes(id) ON DELETE CASCADE,
    tag_id INTEGER NOT NULL REFERENCES tags(id) ON DELETE CASCADE,
    PRIMARY KEY (note_id, tag_id)
  );
  CREATE TABLE segments (
    id INTEGER PRIMARY KEY,
    session_id INTEGER NOT NULL REFERENCES sessions(id) ON DELETE CASCADE,
    t_start INTEGER NOT NULL,
    t_end INTEGER NOT NULL,
    text TEXT NOT NULL
  );
  CREATE TABLE events (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    props_json TEXT NOT NULL DEFAULT '{}',
    created_at INTEGER NOT NULL
  );
  CREATE TABLE settings (
    key TEXT PRIMARY KEY,
    value_json TEXT NOT NULL
  );
  CREATE VIRTUAL TABLE notes_fts USING fts5(
    title, body_md, content='notes', content_rowid='id'
  );
  CREATE TRIGGER notes_ai AFTER INSERT ON notes BEGIN
    INSERT INTO notes_fts(rowid, title, body_md) VALUES (new.id, new.title, new.body_md);
  END;
  CREATE TRIGGER notes_ad AFTER DELETE ON notes BEGIN
    INSERT INTO notes_fts(notes_fts, rowid, title, body_md) VALUES ('delete', old.id, old.title, old.body_md);
  END;
  CREATE TRIGGER notes_au AFTER UPDATE ON notes BEGIN
    INSERT INTO notes_fts(notes_fts, rowid, title, body_md) VALUES ('delete', old.id, old.title, old.body_md);
    INSERT INTO notes_fts(rowid, title, body_md) VALUES (new.id, new.title, new.body_md);
  END;
  `
]

export function openDb(path: string): Database.Database {
  const db = new Database(path)
  db.pragma('journal_mode = WAL')
  db.pragma('foreign_keys = ON')
  migrate(db)
  return db
}

function migrate(db: Database.Database): void {
  const current = db.pragma('user_version', { simple: true }) as number
  for (let i = current; i < MIGRATIONS.length; i++) {
    db.transaction(() => {
      db.exec(MIGRATIONS[i])
      db.pragma(`user_version = ${i + 1}`)
    })()
  }
}
```

```ts
import { existsSync, readFileSync, writeFileSync } from 'node:fs'

type SafeStorageLike = {
  isEncryptionAvailable(): boolean
  encryptString(s: string): Buffer
  decryptString(b: Buffer): string
}

export function createSecretStore(opts: { safeStorage: SafeStorageLike; file: string }) {
  const { safeStorage, file } = opts

  function load(): Record<string, string> {
    if (!existsSync(file)) return {}
    return JSON.parse(readFileSync(file, 'utf8'))
  }

  return {
    set(name: string, value: string): void {
      if (!safeStorage.isEncryptionAvailable()) {
        throw new Error('OS encryption is unavailable; refusing to store the API key')
      }
      const all = load()
      all[name] = safeStorage.encryptString(value).toString('base64')
      writeFileSync(file, JSON.stringify(all, null, 2))
    },
    get(name: string): string | null {
      const all = load()
      if (!(name in all)) return null
      return safeStorage.decryptString(Buffer.from(all[name], 'base64'))
    },
    delete(name: string): void {
      const all = load()
      delete all[name]
      writeFileSync(file, JSON.stringify(all, null, 2))
    }
  }
}
```

```ts
import type Database from 'better-sqlite3'

export type Note = {
  id: number
  title: string
  body_md: string
  folder_id: number | null
  favorite: boolean
  archived: boolean
  session_id: number | null
  created_at: number
  updated_at: number
}

export type Folder = { id: number; name: string; parent_id: number | null }

type Row = Omit<Note, 'favorite' | 'archived'> & { favorite: number; archived: number }

const toNote = (r: Row): Note => ({ ...r, favorite: !!r.favorite, archived: !!r.archived })

// Wrap user text as an FTS5 phrase so operators like " or * are treated literally.
const ftsPhrase = (q: string) => '"' + q.replace(/"/g, '""') + '"'

export function createNotesRepo(db: Database.Database) {
  const now = () => Date.now()

  return {
    create(input: { title?: string; body_md?: string; folder_id?: number | null; session_id?: number | null }): Note {
      const t = now()
      const info = db
        .prepare(
          `INSERT INTO notes (title, body_md, folder_id, session_id, created_at, updated_at)
           VALUES (@title, @body_md, @folder_id, @session_id, @t, @t)`
        )
        .run({
          title: input.title ?? '',
          body_md: input.body_md ?? '',
          folder_id: input.folder_id ?? null,
          session_id: input.session_id ?? null,
          t
        })
      return this.get(Number(info.lastInsertRowid))!
    },

    update(id: number, patch: Partial<Pick<Note, 'title' | 'body_md' | 'folder_id' | 'favorite' | 'archived'>>): Note {
      const current = this.get(id)
      if (!current) throw new Error(`note ${id} not found`)
      const next = { ...current, ...patch }
      db.prepare(
        `UPDATE notes SET title=@title, body_md=@body_md, folder_id=@folder_id,
           favorite=@favorite, archived=@archived, updated_at=@t WHERE id=@id`
      ).run({
        id,
        title: next.title,
        body_md: next.body_md,
        folder_id: next.folder_id,
        favorite: next.favorite ? 1 : 0,
        archived: next.archived ? 1 : 0,
        t: now()
      })
      return this.get(id)!
    },

    get(id: number): Note | null {
      const r = db.prepare('SELECT * FROM notes WHERE id = ?').get(id) as Row | undefined
      return r ? toNote(r) : null
    },

    list(filter: { folder_id?: number | null; favorite?: boolean; archived?: boolean; tag?: string } = {}): Note[] {
      const where: string[] = []
      const args: Record<string, unknown> = {}
      if (filter.folder_id !== undefined) {
        where.push(filter.folder_id === null ? 'folder_id IS NULL' : 'folder_id = @folder_id')
        if (filter.folder_id !== null) args.folder_id = filter.folder_id
      }
      if (filter.favorite !== undefined) { where.push('favorite = @favorite'); args.favorite = filter.favorite ? 1 : 0 }
      if (filter.archived !== undefined) { where.push('archived = @archived'); args.archived = filter.archived ? 1 : 0 }
      if (filter.tag !== undefined) {
        where.push('id IN (SELECT nt.note_id FROM note_tags nt JOIN tags t ON t.id = nt.tag_id WHERE t.name = @tag)')
        args.tag = filter.tag
      }
      const sql = `SELECT * FROM notes ${where.length ? 'WHERE ' + where.join(' AND ') : ''} ORDER BY updated_at DESC`
      return (db.prepare(sql).all(args) as Row[]).map(toNote)
    },

    search(query: string): Note[] {
      const q = query.trim()
      if (!q) return []
      const rows = db
        .prepare(
          `SELECT n.* FROM notes_fts f JOIN notes n ON n.id = f.rowid
           WHERE notes_fts MATCH ? ORDER BY rank`
        )
        .all(ftsPhrase(q)) as Row[]
      return rows.map(toNote)
    },

    setTags(noteId: number, names: string[]): void {
      const clean = [...new Set(names.map((n) => n.trim()).filter(Boolean))]
      db.transaction(() => {
        db.prepare('DELETE FROM note_tags WHERE note_id = ?').run(noteId)
        for (const name of clean) {
          db.prepare('INSERT OR IGNORE INTO tags (name) VALUES (?)').run(name)
          const tag = db.prepare('SELECT id FROM tags WHERE name = ?').get(name) as { id: number }
          db.prepare('INSERT INTO note_tags (note_id, tag_id) VALUES (?, ?)').run(noteId, tag.id)
        }
      })()
    },

    tagsFor(noteId: number): string[] {
      return (db
        .prepare('SELECT t.name FROM tags t JOIN note_tags nt ON nt.tag_id = t.id WHERE nt.note_id = ? ORDER BY t.name')
        .all(noteId) as { name: string }[]).map((r) => r.name)
    },

    createFolder(name: string, parentId: number | null = null): Folder {
      const info = db.prepare('INSERT INTO folders (name, parent_id) VALUES (?, ?)').run(name, parentId)
      return db.prepare('SELECT * FROM folders WHERE id = ?').get(Number(info.lastInsertRowid)) as Folder
    }
  }
}
```

```ts
const RESERVED = /^(con|prn|aux|nul|com[0-9]|lpt[0-9])$/i

export function noteToMarkdown(note: Note, tags: string[]): string {
  const title = note.title.replace(/\\/g, '\\\\').replace(/"/g, '\\"')
  const tagList = tags.length ? `[${tags.join(', ')}]` : '[]'
  return `---\ntitle: "${title}"\ntags: ${tagList}\ncreated: ${new Date(note.created_at).toISOString()}\n---\n\n${note.body_md}\n`
}

export function safeFilename(title: string, id: number): string {
  const cleaned = title
    .replace(/[\\/:*?"<>|\x00-\x1f]/g, '')
    .replace(/\s+/g, ' ')
    .trim()
    .slice(0, 100)
    .replace(/[. ]+$/, '')
  if (!cleaned || RESERVED.test(cleaned)) return `note-${id}.md`
  return `${cleaned}.md`
}
```

```ts
  role: 'system' | 'user' | 'assistant'

  model: string
  messages: ChatMessage[]

  id: string
  complete(req: ChatRequest): Promise<string>
  stream(req: ChatRequest): AsyncIterable<string>

export class ProviderError extends Error {
  constructor(
    public kind: 'auth' | 'rate_limit' | 'timeout' | 'unsupported' | 'other',
    message: string,
    public status?: number
  ) {
    super(message)
    this.name = 'ProviderError'
  }
}

export function errorFromStatus(status: number, body: string): ProviderError {
  if (status === 401 || status === 403) return new ProviderError('auth', `Authentication failed (${status})`, status)
  if (status === 429) return new ProviderError('rate_limit', 'Rate limit reached', status)
  if (status === 400 && /image|vision|multimodal/i.test(body)) {

  }
  return new ProviderError('other', `Provider error ${status}: ${body.slice(0, 200)}`, status)
```

```ts
import { ChatMessage, ChatRequest, LLMProvider, ProviderError, errorFromStatus } from './types'

type Opts = { id: string; baseUrl: string; apiKey: string | null; fetch?: typeof fetch }

function toWire(messages: ChatMessage[]) {
  return messages.map((m) => {
    if (typeof m.content === 'string') return { role: m.role, content: m.content }
    return {
      role: m.role,
      content: m.content.map((p) =>
        p.type === 'text'
          ? { type: 'text', text: p.text }
          : { type: 'image_url', image_url: { url: `data:${p.mime};base64,${p.base64}` } }
      )
    }
  })
}

export function createOpenAICompatProvider(opts: Opts): LLMProvider {
  const f = opts.fetch ?? fetch
  const url = opts.baseUrl.replace(/\/$/, '') + '/chat/completions'

  function headers(): Record<string, string> {
    const h: Record<string, string> = { 'content-type': 'application/json' }
    if (opts.apiKey) h.Authorization = `Bearer ${opts.apiKey}`
    return h
  }

  async function send(req: ChatRequest, stream: boolean): Promise<Response> {
    let res: Response
    try {
      res = await f(url, {
        method: 'POST',
        headers: headers(),
        body: JSON.stringify({ model: req.model, messages: toWire(req.messages), max_tokens: req.maxTokens, stream }),
        signal: req.signal
      })
    } catch (e: any) {
      if (e?.name === 'TimeoutError' || e?.name === 'AbortError') throw new ProviderError('timeout', 'Request timed out')
      throw new ProviderError('other', String(e?.message ?? e))
    }
    if (!res.ok) throw errorFromStatus(res.status, await res.text())
    return res
  }

  return {
    id: opts.id,
    async complete(req) {
      const res = await send(req, false)
      const data = (await res.json()) as { choices: { message: { content: string } }[] }
      return data.choices[0].message.content
    },
    async *stream(req) {
      const res = await send(req, true)
      const reader = res.body!.getReader()
      const dec = new TextDecoder()
      let buf = ''
      while (true) {
        const { value, done } = await reader.read()
        if (done) break
        buf += dec.decode(value, { stream: true })
        let idx: number
        while ((idx = buf.indexOf('\n\n')) !== -1) {
          const event = buf.slice(0, idx)
          buf = buf.slice(idx + 2)
          for (const line of event.split('\n')) {
            if (!line.startsWith('data: ')) continue
            const payload = line.slice(6).trim()
            if (payload === '[DONE]') return
            const delta = JSON.parse(payload).choices?.[0]?.delta?.content
            if (delta) yield delta as string
          }
        }
      }
    }
  }
}
```

```ts
import { ChatMessage, ChatRequest, ContentPart, LLMProvider, ProviderError, errorFromStatus } from './types'

type Opts = { apiKey: string; baseUrl?: string; fetch?: typeof fetch }

function toBlocks(parts: ContentPart[]) {
  return parts.map((p) =>
    p.type === 'text'
      ? { type: 'text', text: p.text }
      : { type: 'image', source: { type: 'base64', media_type: p.mime, data: p.base64 } }
  )
}

function split(messages: ChatMessage[]) {
  const system = messages
    .filter((m) => m.role === 'system')
    .map((m) => (typeof m.content === 'string' ? m.content : m.content.map((p) => (p.type === 'text' ? p.text : '')).join('')))
    .join('\n')
  const rest = messages
    .filter((m) => m.role !== 'system')
    .map((m) => ({ role: m.role, content: typeof m.content === 'string' ? m.content : toBlocks(m.content) }))
  return { system, rest }
}

export function createAnthropicProvider(opts: Opts): LLMProvider {
  const f = opts.fetch ?? fetch
  const url = (opts.baseUrl ?? 'https://api.anthropic.com') + '/v1/messages'

  async function send(req: ChatRequest, stream: boolean): Promise<Response> {
    const { system, rest } = split(req.messages)
    let res: Response
    try {
      res = await f(url, {
        method: 'POST',
        headers: {
          'content-type': 'application/json',
          'x-api-key': opts.apiKey,
          'anthropic-version': '2023-06-01'
        },
        body: JSON.stringify({
          model: req.model,
          max_tokens: req.maxTokens ?? 1024,
          system: system || undefined,
          messages: rest,
          stream
        }),
        signal: req.signal
      })
    } catch (e: any) {
      if (e?.name === 'TimeoutError' || e?.name === 'AbortError') throw new ProviderError('timeout', 'Request timed out')
      throw new ProviderError('other', String(e?.message ?? e))
    }
    if (!res.ok) throw errorFromStatus(res.status, await res.text())
    return res
  }

  return {
    id: 'anthropic',
    async complete(req) {
      const data = (await (await send(req, false)).json()) as { content: { type: string; text?: string }[] }
      return data.content.filter((b) => b.type === 'text').map((b) => b.text).join('')
    },
    async *stream(req) {
      const res = await send(req, true)
      const reader = res.body!.getReader()
      const dec = new TextDecoder()
      let buf = ''
      while (true) {
        const { value, done } = await reader.read()
        if (done) break
        buf += dec.decode(value, { stream: true })

            if (evt.type === 'content_block_delta' && evt.delta?.type === 'text_delta') yield evt.delta.text as string
          }
        }
      }
    }
  }
```

```ts
import { LLMProvider, ChatRequest, ProviderError } from './types'

export type ProviderConfig = {
  providerId: string
  model: string
  needsKey: boolean
  baseUrl: string
}

type Deps = {
  secrets: { get(name: string): string | null }
  build(config: ProviderConfig, apiKey: string | null): LLMProvider
  sleep: (ms: number) => Promise<void>
}

const BACKOFF_MS = [500, 1000, 2000]

export function createRegistry(deps: Deps) {
  let config: ProviderConfig | null = null
  let killed = false

  function resolve(): LLMProvider {
    if (killed) throw new ProviderError('other', 'Network is disabled by the kill switch')
    if (!config) throw new ProviderError('other', 'No provider configured')
    let apiKey: string | null = null
    if (config.needsKey) {
      apiKey = (deps.secrets.get(config.providerId) ?? '').trim()
      if (!apiKey) throw new ProviderError('auth', `No API key set for ${config.providerId}`)
    }
    return deps.build(config, apiKey)
  }

  async function withRetry<T>(fn: () => Promise<T>): Promise<T> {
    for (let attempt = 0; ; attempt++) {
      try {
        return await fn()
      } catch (e) {
        const retryable = e instanceof ProviderError && e.kind === 'rate_limit'
        if (!retryable || attempt >= BACKOFF_MS.length) throw e
        await deps.sleep(BACKOFF_MS[attempt])
      }
    }
  }

  return {
    configure(c: ProviderConfig) {
      config = c
    },
    setKillSwitch(on: boolean) {
      killed = on
    },
    call(req: Omit<ChatRequest, 'model'>): Promise<string> {
      return withRetry(() => {
        const p = resolve()
        return p.complete({ ...req, model: config!.model })
      })
    }
  }
}
```

```ts
import { errorFromStatus, ProviderError } from '../providers/types'
import type { Transcriber } from './types'

type Opts = { apiKey: string; model: string; baseUrl?: string; fetch?: typeof fetch }

export function createOpenAITranscriber(opts: Opts): Transcriber {
  const f = opts.fetch ?? fetch
  const url = (opts.baseUrl ?? 'https://api.openai.com/v1') + '/audio/transcriptions'

  return {
    async transcribe(audio, o) {
      const form = new FormData()
      form.append('file', new Blob([audio.bytes], { type: audio.mime }), 'chunk.webm')
      form.append('model', opts.model)
      if (o?.language) form.append('language', o.language)

      let res: Response
      try {
        res = await f(url, { method: 'POST', headers: { Authorization: `Bearer ${opts.apiKey}` }, body: form })
      } catch (e: any) {
        throw new ProviderError('other', String(e?.message ?? e))
      }
      if (!res.ok) throw errorFromStatus(res.status, await res.text())
      const data = (await res.json()) as { text: string }
      return { text: data.text.trim() }
    }
  }
}
```

```ts
export type CaptureHandle = { stop(): void }

const CHUNK_MS = 5000

export async function startCapture(opts: {
  onChunk(bytes: Uint8Array, mime: string, startMs: number, endMs: number): void
  onDeviceLost(): void
}): Promise<CaptureHandle> {
  const mic = await navigator.mediaDevices.getUserMedia({ audio: true })
  const display = await navigator.mediaDevices.getDisplayMedia({ video: true, audio: true })
  const sysTracks = display.getAudioTracks()
  display.getVideoTracks().forEach((t) => t.stop())

  // Mix mic and system audio into one stream the recorder can read.
  const ctx = new AudioContext()
  const dest = ctx.createMediaStreamDestination()
  ctx.createMediaStreamSource(mic).connect(dest)
  if (sysTracks.length) ctx.createMediaStreamSource(new MediaStream(sysTracks)).connect(dest)

  const mime = 'audio/webm;codecs=opus'
  const recorder = new MediaRecorder(dest.stream, { mimeType: mime })
  const started = performance.now()
  let lastStart = 0

  recorder.ondataavailable = async (e) => {
    if (!e.data.size) return
    const now = performance.now() - started
    const bytes = new Uint8Array(await e.data.arrayBuffer())
    opts.onChunk(bytes, mime, lastStart, now)
    lastStart = now
  }

  const deviceEnded = () => opts.onDeviceLost()
  mic.getAudioTracks().forEach((t) => t.addEventListener('ended', deviceEnded))
  sysTracks.forEach((t) => t.addEventListener('ended', deviceEnded))

  recorder.start(CHUNK_MS)

  return {
    stop() {
      if (recorder.state !== 'inactive') recorder.stop()
      mic.getTracks().forEach((t) => t.stop())
      display.getTracks().forEach((t) => t.stop())
      ctx.close()
    }
  }
}
```

```ts
import type { Transcriber } from '../transcription/types'

export type Segment = { t_start: number; t_end: number; text: string }

type Deps = {
  transcriber: Transcriber
  onSegment(seg: Segment): void
  onError(err: unknown): void
  saveSession(segments: Segment[]): number
}

export function createListenLoop(deps: Deps) {
  let active = false
  const segments: Segment[] = []

  return {
    start() {
      active = true
      segments.length = 0
    },
    async pushChunk(bytes: Uint8Array, mime: string, startMs: number, endMs: number): Promise<void> {
      if (!active) return
      try {
        const { text } = await deps.transcriber.transcribe({ bytes, mime })
        if (!text) return
        const seg = { t_start: startMs, t_end: endMs, text }
        segments.push(seg)
        deps.onSegment(seg)
      } catch (e) {
        deps.onError(e)
      }
    },
    stop(_reason: 'user' | 'device_lost'): { sessionId: number } {
      active = false
      const sessionId = deps.saveSession([...segments])
      return { sessionId }
    }
  }
}
```

```ts
import type { Segment } from './listen'

const OPENERS = [
  'what', 'why', 'how', 'when', 'where', 'who', 'which', 'can', 'could', 'would', 'will',
  'do', 'does', 'did', 'is', 'are', 'tell', 'explain', 'describe', 'walk', 'give',

  'كيف', 'ماذا', 'لماذا', 'هل', 'من', 'أين', 'متى'
]

export function isQuestion(text: string): boolean {
  const t = text.trim()
  if (!t) return false
  if (/[?؟]\s*$/.test(t)) return true
  const first = t.toLowerCase().split(/\s+/)[0].replace(/[^\p{L}]/gu, '')
  return OPENERS.includes(first)
}

export function detectQuestions(segments: Segment[]): Segment[] {
  return segments.filter((s) => isQuestion(s.text))
}
```

```ts
import type { ChatRequest } from '../providers/types'
import type { Segment } from './listen'

type Chat = { stream(req: ChatRequest): AsyncIterable<string> }

const SYSTEM =
  'You help the user answer questions in a live conversation. Give a concise, direct answer they can say out loud. Keep it under 120 words unless asked otherwise.'

export function createSuggester(deps: { chat: Chat; contextSegments?: number }) {
  const n = deps.contextSegments ?? 20

  function contextBlock(recent: Segment[] | null): string {
    if (!recent || recent.length === 0) return ''
    const last = recent.slice(-n)
    return 'Recent conversation:\n' + last.map((s) => `- ${s.text}`).join('\n') + '\n\n'
  }

  async function run(userText: string, onToken: (t: string) => void): Promise<string> {
    let full = ''
    for await (const t of deps.chat.stream({
      model: '', // model is filled in by the registry
      messages: [
        { role: 'system', content: SYSTEM },
        { role: 'user', content: userText }
      ]
    })) {
      full += t
      onToken(t)
    }
    return full
  }

  return {
    suggest(question: string, recent: Segment[], onToken: (t: string) => void) {
      return run(`${contextBlock(recent)}Question: ${question}`, onToken)
    },
    ask(question: string, recent: Segment[] | null, onToken: (t: string) => void) {
      return run(`${contextBlock(recent)}${question}`, onToken)
    }
  }
}
```

```ts
import type Database from 'better-sqlite3'

const CONSENT_KEY = 'tracking.consent'

export function createTracker(db: Database.Database, opts: { now?: () => number } = {}) {
  const now = opts.now ?? (() => Date.now())

  const getSetting = (key: string): unknown => {
    const r = db.prepare('SELECT value_json FROM settings WHERE key = ?').get(key) as { value_json: string } | undefined
    return r ? JSON.parse(r.value_json) : undefined
  }
  const setSetting = (key: string, value: unknown) =>
    db.prepare('INSERT INTO settings (key, value_json) VALUES (?, ?) ON CONFLICT(key) DO UPDATE SET value_json = excluded.value_json').run(key, JSON.stringify(value))

  return {
    setConsent(granted: boolean) {
      setSetting(CONSENT_KEY, granted)
    },
    hasConsent(): boolean {
      return getSetting(CONSENT_KEY) === true
    },
    record(name: string, props: Record<string, unknown> = {}) {
      if (!this.hasConsent()) return
      db.prepare('INSERT INTO events (name, props_json, created_at) VALUES (?, ?, ?)').run(name, JSON.stringify(props), now())
    },
    activityLog(limit = 200) {
      return (db
        .prepare('SELECT name, props_json, created_at FROM events ORDER BY id DESC LIMIT ?')
        .all(limit) as { name: string; props_json: string; created_at: number }[]).map((r) => ({
        name: r.name,
        props: JSON.parse(r.props_json),
        created_at: r.created_at
      }))
    },
    revokeAndWipe() {
      db.transaction(() => {
        db.prepare('DELETE FROM events').run()
        setSetting(CONSENT_KEY, false)
      })()
    }
  }
}
```

```tsx
import { useEffect, useState } from 'react'

type Entry = { name: string; props: Record<string, unknown>; created_at: number }

export function ActivityView() {
  const [consent, setConsent] = useState<boolean | null>(null)
  const [log, setLog] = useState<Entry[]>([])

  useEffect(() => {
    window.miny.settings.get('tracking.consent').then((v: any) => setConsent(v === true))
    window.miny.tracking.log().then(setLog)
  }, [])

  const grant = async (on: boolean) => {
    await window.miny.tracking.setConsent(on)
    setConsent(on)
  }

  return (
    <div>
      <h2>Activity log</h2>
      {consent === false && (
        <div>
          <p>Miny can record usage events on this device to show here. Nothing is recorded until you allow it.</p>
          <button onClick={() => grant(true)}>Allow recording</button>
        </div>
      )}
      {consent && (
        <>
          <button onClick={async () => { await window.miny.tracking.wipe(); setConsent(false); setLog([]) }}>Turn off and delete history</button>
          <ul>
            {log.map((e, i) => (
              <li key={i}>{new Date(e.created_at).toLocaleString()}: {e.name} {JSON.stringify(e.props)}</li>
            ))}
          </ul>
        </>
      )}
    </div>
  )
}
```
